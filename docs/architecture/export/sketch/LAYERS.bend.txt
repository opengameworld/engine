# The engine's layers, and which may import which. (SKETCH, revision 2: lives at the repo
# root beside PROJECT.bend. Checked on EVERY `bun run check` (tools/check.ts fence step) and
# on EVERY export (export/contract.bend: Forbidden, against what the bundler really read).)
#
# One table, places(), is the only list of folders in the engine. Everything else is derived
# from it: layer_of(path) (the first place whose prefix matches), the folders check.ts
# type-checks (typed_roots(), replacing check.ts's hand-written FOLDERS), and what ships.
# A new top-level folder is ONE line here; forgetting it makes its files Outside, which the
# fence refuses, so the folder cannot be silently unchecked.
#
#   places()            the table: prefix, layer, type-checked?
#   layer_of(path)      which layer a repo path is in
#   may_import(a, b)    may a file of layer a import a file of layer b?
#   ships(l)            may a file of layer l be inside a published game?
#   debts()             edges allowed for now although they break a rule, and why
#
# Paths are repo-relative. In a CREATOR's repo the engine is a dependency: a path under
# node_modules/ogw-engine/ is judged as the same path in the engine (layer_of strips the
# prefix), and the creator's own files (GAME.bend and its folders) are Project.
#
# Limits now:
# - Edges are found by text (`import ./x.bend`, `from "x"`, `src="x"`), as the refs/ fence
#   does today; export closes the gap with the bundler's real inputs.
# - engine/world.bend and engine/gizmo.bend import Bend-UI: three named debts (moving gizmo
#   is Isabella's call). Each debt that no longer exists fails the check (the list cannot rot).
# - platform/web/materials.js re-validates shader values in JS (a known debt of the rule
#   "decisions in Bend"); not an import edge, so not listed here.
#
# `TODO` marks a body not written yet (sketch).

type Layer is Data:
  Engine{}            # engine/**                         pure Bend runtime rules
  PlatWeb{}           # platform/web/**                    JS that reads input and draws
  PlatNative{}        # platform/native/**                 later
  PlatServer{}        # platform/server/**                 the world server's shell, later
  Server{}            # server/**                          the world's rules on a server, later
  Project{}           # projects/**  (and a creator's GAME.bend + folders)
  Catalog{}           # PROJECT.bend                       the list of projects; the game entry resolves its project here
  Mannequin{}         # mannequin/scene.bend, credit.bend, gen/{skeleton,rules}.bend, gen/skin.*, gen/*.png
  MannequinLaws{}     # mannequin/LAWS.bend, PROOF.bend, gen/weights.bend (141 KB, laws only)
  MannequinSource{}   # mannequin/source/, gltf/, rig.json, CREDITS.md (4.2 MB of originals)
  Library{}           # library/<c>/gen/m/** (one motion a file), library/<c>/gen/index.bend
  LibraryIndex{}      # library/catalog.bend, library/<c>/gen/all.bend, gen/thumbs.bend (the editor's)
  LibrarySource{}     # library/<c>/source/, clips.json, CREDITS.md
  GameGen{}           # out/gen/<game>/**                  written by export for one game
  GameUi{}            # ui/**                              Bend-UI pages export renders to static text (credits)
  UiKit{}             # node_modules/bend-ui/**
  Editor{}            # editor/** (workspace/, publish/, shaders/ generated), index.html, main.js, KEYS.bend
  Contract{}          # export/contract.bend, LAYERS.bend   the folder's rules (read by export, publish, editor)
  Export{}            # export/**                          the build tool
  Publish{}           # publish/**                         the publish machine's rules (pure Bend)
  PlatPublish{}       # platform/publish/**                the publish server's shell (Bun), runs on the VPS
  Tools{}             # tools/** (tools/shaders/ too)
  Tests{}             # tests/**, LAWS.bend, PROOF.bend, SPEC.bend
  Refs{}              # refs/**
  Outside{}           # a path places() does not know: the fence refuses it

  Place{prefix: String, layer: Layer, typed: Bool}   # typed: check.ts runs --check-only on its .bend

def ships(l: Layer) -> Bool:
  match l:
    case Engine{}:          True{}
    case PlatWeb{}:         True{}
    case PlatNative{}:      True{}
    case Project{}:         True{}
    case Mannequin{}:       True{}     # with its credit (contract.bend credit gate)
    case Library{}:         True{}     # only the motions a game imports: out/gen/<game>/clips.bend
    case GameGen{}:         True{}
    case GameUi{}:          False{}    # rendered by export to static text (site/credits.html); no script ships
    case UiKit{}:           True{}     # through the named debts only; license unknown (export/credit.bend bend_ui)
    case Catalog{}:         True{}     # ~28 KB gz for every project today; a creator's repo has one (GAME.bend, later)
    case MannequinLaws{}:   False{}
    case MannequinSource{}: False{}
    case LibraryIndex{}:    False{}    # thumbs and the 42-clip index are the editor's
    case LibrarySource{}:   False{}
    case PlatServer{}:      False{}
    case Server{}:          False{}
    case Editor{}:          False{}
    case Contract{}:        False{}
    case Export{}:          False{}
    case Publish{}:         False{}
    case PlatPublish{}:     False{}    # runs on the publish server, never inside a game
    case Tools{}:           False{}
    case Tests{}:           False{}
    case Refs{}:            False{}
    case Outside{}:         False{}
