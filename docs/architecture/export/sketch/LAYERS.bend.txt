# The engine's layers, and which may import which. (SKETCH: lives at the repo root, beside
# PROJECT.bend.) One list of rules; tools/check.ts's fence step and export/ both ask it, so
# "the runtime never imports the editor" and "nothing reaches refs/" are checked on every
# `bun run check` and on every export, not kept by convention.
#
#   layer_of(path)        which layer a repo path is in (paths relative to the repo root)
#   may_import(a, b)      may a file of layer a import a file of layer b?
#   ships(l)              may a file of layer l be inside a published game?
#
# The shell (tools/check.ts) reads files and finds import edges (IO); this file decides.
#
# Limits now:
# - Edges are found by text (`import ./x.bend`, `from "x"`, `src="x"`), as the refs/ fence
#   does today; a dynamic import(`${x}`) is not seen. Export closes the gap: it also checks the
#   inputs the bundler actually read (export/contract.bend, Forbidden).
# - engine/world.bend and engine/gizmo.bend import Bend-UI tokens (colors, motion). Engine -> UiKit
#   is NOT allowed; those three exact edges are listed in debts() (graft from candidate 2). The
#   check prints them every run, and a debt that no longer exists fails the check, so the list
#   cannot rot. Moving gizmo to editor/ is Isabella's call.
#
# `TODO` marks a body not written yet (sketch).

  Engine{}          # engine/**            pure Bend rules: math, scene, input/, tick, play, shader, place
  PlatWeb{}         # platform/web/**      JS that reads input and draws (WebGL2, WebAudio)
  PlatNative{}      # platform/native/**   Bend effect files + C for the native track (later)
  PlatServer{}      # platform/server/**   Bun shell of the world server (later)
  Server{}          # server/**            pure Bend rules of the world server (later)
  Project{}         # projects/**          a game's own content and rules
  Catalog{}         # PROJECT.bend          the list of projects
  UiKit{}           # node_modules/bend-ui/**
  Editor{}          # editor/**, index.html, main.js, KEYS.bend (the editor's page)
  Export{}          # export/**             a tool: reads a project, writes a game folder
  Tools{}           # tools/**
  Tests{}           # tests/**, LAWS.bend, PROOF.bend, SPEC.bend
  Refs{}            # refs/**               third-party references; never ship
  Outside{}         # anything else (a path this file does not know: refused by the fence)

# Whether a layer may be inside a published game (web or native). Everything a game runs on.
def ships(l: Layer) -> Bool:
  match l:
    case Engine{}:     True{}
    case PlatWeb{}:    True{}
    case PlatNative{}: True{}
    case Project{}:    True{}
    case Catalog{}:    True{}
    case UiKit{}:      True{}     # game menus are Bend-UI too (team rule: UI only Bend-UI)
    case PlatServer{}: False{}    # ships to the server, never inside the client folder
    case Server{}:     False{}
    case Editor{}:     False{}
    case Export{}:     False{}
    case Tools{}:      False{}
    case Tests{}:      False{}
    case Refs{}:       False{}
    case Outside{}:    False{}
