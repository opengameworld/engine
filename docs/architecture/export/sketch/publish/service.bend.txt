# The publish server's decisions: every verb of the protocol as a pure function from what the
# server read for this request to what it must do. (SKETCH, new in v2)
#
# publish/server.ts is the hands: HTTP, disk, hashes, symlinks, the log. For each request it reads
# what Bend asks for (a Seen value), calls one function here, and carries out the Plan it gets back,
# step by step, in order (effects as data, CATCHES X6). The same file decides for the team's VPS and
# for the multi-creator service; Mode is the only difference, and it is a value in the server's
# config, not a second code path.
#
# Storage (one owner: the server; every path below comes from a def here, the hands never build one):
#   /srv/ogw/blobs/<sha[0..2]>/<sha>                       content-addressed files, written tmp+rename, never changed
#   /srv/ogw/games/<creator>/<game>/releases/<id>.json     game.json + the Release record
#   /srv/ogw/games/<creator>/<game>/releases/<id>/site/    hard links into blobs/: the served tree, made once
#   /srv/ogw/games/<creator>/<game>/channels/<name>.log    append-only moves; the pointer is the last line
#   /srv/ogw/sites/<label> -> ../games/.../releases/<id>/site   Caddy's view: one symlink per link label,
#                                                          rebuilt from the logs at start (a cache, never the truth)
#   /srv/ogw/keys/<id>.json                                Key + sha256(secret)
#   /srv/ogw/creators/<creator>.json                       plan, key version, people (+ point secret hashes on the VPS)
#   /srv/ogw/log/<yyyy-mm>.jsonl                           every accepted write: who, what, when (from Coolify's
#                                                          deployment rows and the set-aside "log of steps")
#
# Shared state, and what happens when two writers race (separate-before-serializing):
#   blobs     content-addressed: two identical writes write the same bytes; nothing to merge.
#   releases  id = hash of content: creating one twice returns the first (created: false).
#   channels  the only shared mutable thing. A move carries `expect` (the pointer the client saw);
#             the hands hold the channel's lock (flock on the .log) across read-decide-append, so the
#             second of two racing moves gets Stale{now} instead of silently winning.
#   sites/    derived from channels/ and releases/; a crash between appending a move and swapping the
#             symlink is repaired by the next start (reconcile) or the next move.
#
# Limits now:
# - One machine. No replication; the backup is a copy of /srv/ogw (blobs are immutable: rsync is enough).
# - Blobs are never collected (disk is cheap next to a lost release). retain() comes with quotas.
# - isolation (COOP/COEP) is refused: Caddy's config is the same for every site (Limits.isolation).

  Team{creator: String}        # the VPS: one creator, person secrets set on the box, no login, no approvals

type Plan is Data:

  Free{}

type Limits is Data:
  Limits{files: U32, file_bytes: U32, release_bytes: U32, store_bytes: U32, releases_a_day: U32, isolation: Bool}

# ---------- What the hands read, per verb ----------

type Caller is Data:
  Caller{key: K.PubKey, creator_version: U32, plan: Plan, today: U32, now: U32}

# A blob as the server measured it on receipt (never as the client said).
type Blob is Data:
  Blob{sha: String, bytes: U32, gz: U32}

type SeenRelease is Data:
  SeenRelease{
    game: String,
    manifest: C.Manifest,          # parsed from game.json by the hands, entries' sizes REPLACED by blob facts
    id: String,                    # sha256(C.id_text(manifest)) computed by the hands on the server
    missing: List<&2, String>,     # entry shas the store lacks
    existing: Maybe<&2, Pr.Release>,  # a release with this id already
    known: List<&2, C.Work>,       # works the store recognises by blob sha (known_works below)
    live: Maybe<&2, String>,       # the game's live pointer, for Release.parent
    used_bytes: U32,               # the creator's store use, for quotas
    made_today: U32
  }

type SeenMove is Data:
  SeenMove{
    to: Pr.Address,
    id: String,
    release: Maybe<&2, Pr.Release>,
    moves: List<&2, Pr.Shift>,      # the channel's log, newest first
    expect: Maybe<&2, String>,     # what the client saw as the pointer
    presence: K.Presence
  }

# ---------- What the server answers and does ----------

type Denial is Data:
  ByKey{why: K.Denied}
  Upgrade{min: U32, max: U32}
  BadBlob{sha: String}             # body's sha256 is not the sha in the path
  MissingBlobs{shas: List<&2, String>}
  Contract{reasons: List<&2, C.Reason>}   # C.recheck(manifest) on the server's own measurements
  Uncredited{work: C.Work}         # a known work's blob ships and the manifest lacks its credit
  OverQuota{what: String}
  IdTaken{id: String}              # same 12 hex, different content: refused, never overwritten
  NoSuchRelease{id: String}
  Unchecked{}                      # source.checked False: never on a guarded channel
  Dirty{}                          # source.dirty True: never on a guarded channel
  SeedChanged{live: String, new: String}  # AGENTS.md: the world's seed never changes once live
  Stale{now: Maybe<&2, String>}    # the pointer moved since the client looked
  IsolationNotServed{}

type Step is Data:
  WriteRelease{release: Pr.Release}          # releases/<id>.json, then the site/ link farm
  AppendMove{to: Pr.Address, shift: Pr.Shift}  # channels/<name>.log
  PointSite{label: String, id: String}       # sites/<label> -> that release's site/ (atomic: ln -s tmp + rename)
  Log{line: String}

type Answer is Data:
  Apply{steps: List<&2, Step>}               # not `Done`: a Base constructor (CATCHES N1)
  NoChange{}                                 # idempotent repeat: 200 with what is already there
  Approve{}                                  # service, guarded move, no presence yet: hands open an approval (202)
  Refuse{why: List<&2, Denial>}
