# export/LAWS.bend: the laws of the game folder. Proved in export/PROOF.bend, run by tools/check.ts
# step "laws" (one more `bend export/PROOF.bend` line, like editor/defaults and mannequin). (SKETCH)
#
# Every law here is shaped to reduce (CATCHES F1, F6, M1): the decisive parameter is matched first,
# so the right side is reached by reduction whatever the other arguments are. Facts that need a
# walk over Strings (layers of paths, budgets, ids) are numeric tests in tests/export.bend.

import Base
import ./contract.bend as C

# Nothing ships if `bun run check` failed.
law nothing_ships_when_checks_fail:
  for f: C.Facts {C.release(False{}, f) == C.Refused{[C.ChecksFailed{}]} : C.Outcome}

# The mannequin (CC BY 4.0) never ships with its credit hidden: whatever else the game ships.
law mannequin_needs_its_credit:
  for ws: List<&2, C.Work>
  {C.attribution(False{}, C.Mannequin{} <> ws) == C.Uncredited{C.Mannequin{}} <> C.attribution(False{}, ws) : List<&2, C.Reason>}

# Its credit travels: a manifest's credits are derived from its works, so the mannequin's is first
# wherever the mannequin is first.
law mannequin_credit_travels:
  for ws: List<&2, C.Work>
  {C.credits(C.Mannequin{} <> ws) == C.credit(C.Mannequin{}) <> C.credits(ws) : List<&2, C.Credit>}

# Credits shown on the Credits page refuse nothing.
law shown_credits_refuse_nothing:
  for ws: List<&2, C.Work> {C.attribution(C.InMenu{}, ws) == [] : List<&2, C.Reason>}

law page_is_revalidated: {C.cache(C.Page{}) == C.Revalidate{} : C.Cache}
law code_is_immutable:   {C.cache(C.Code{}) == C.Immutable{} : C.Cache}
law assets_are_immutable: {C.cache(C.Asset{}) == C.Immutable{} : C.Cache}

#   - a Facts with "mannequin/gen/skin.bin" in inputs ships credits [credit(Mannequin{})] and the
#     warning NotClearedForSale; with no credits page it is Refused{[Uncredited{Mannequin{}}]}

#   - id_text is the same for the same entries given in two orders, and changes with seed
#   - missing(m, entries(m)) == []
#   - recheck(manifest(f)) == [] for every Facts that release ships (server and client agree)
