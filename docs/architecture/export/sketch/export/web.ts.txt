// export/web.ts: project -> game folder (SKETCH). The thin shell around export/contract.bend:
// it runs things and measures files (IO); every decision is a call into Bend.
//
//   exportWeb({ project })  -> { ok: true, folder } | { ok: false, reasons }
//   verifyFolder(dir)       -> { ok: true, folder } | { ok: false, missing | reasons }
//   serveFolder(dir, port)  -> a local server that sends contract.headers() per kind
//
// Crash- and rerun-safe: everything is written to out/web/.stage-<pid>/, game.json last, then
// the stage is renamed over out/web/<game>/ in one step. A folder without game.json is
// unfinished and every reader refuses it. Same sources -> same build id (contract.id_text), so
// running it twice gives the same folder and every host sees nothing new.
//
// Limits now:
// - Runs `bun run check` in full each time (tens of seconds is fine for a release; previews
//   may later reuse a check result keyed by the work tree's hash).
// - The bundle's inputs come from Bun.build's metafile [U: Bun.build `metafile: true` exists in
//   the pinned Bun; if not, the inputs are found by walking imports from the page, as the fence
//   does]. Either way LAYERS.bend decides.

// The folder as the rest of export/ sees it: parsed once at the boundary (game.json -> Bend
// Manifest value), never re-read as loose JSON by adapters.
export type GameFolder = {
  dir: string;           // out/web/<game>
  site: string;          // out/web/<game>/site  (what is served)
  buildId: string;       // 12 hex, from Contract.id_text

};
type Manifest = { readonly $: "Manifest" };   // opaque: only Bend reads its fields
