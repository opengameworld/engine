# The game folder: the one shape every export writes and every publish host stores. (SKETCH, v2)
#
# Revised from the approved v1. What survives unchanged: the fixed folder, `release(checks, facts)`
# with `checks` matched first, Role (not Kind), Target, Source, seed, isolation, headers and cache
# from Bend, the build id as sha256 of a canonical text. What v2 adds: Asset entries in format 1,
# a budget that counts Code and Assets apart (numbers from a reach rule, measured sizes below),
# credits of what ships with the mannequin's law, the build machine's toolchain, and the clips a
# game takes.
#
#   out/web/<game>/
#     game.json                    the manifest (format 1), written LAST: a folder without it is unfinished
#     site/                        exactly what is served, rooted at /
#       index.html                 the one page (Page): revalidated on every load
#       _/<name>-<hash>.js|css     bundle (Code): immutable
#       a/<hash>.<ext>             files the bundle loads (Asset: skin.bin, textures, shader catalog): immutable
#
#   release(checks, facts) -> Shipped{manifest} | Refused{reasons}
#   recheck(m)                -> the reasons a manifest alone shows (the publish server runs it: it never trusts a client)
#   cache(role), headers(isolation, role)
#   id_text(m)                -> the text the shell hashes into the 12-hex build id (BendHub's own rule, below)
#   json(m, id), missing(m, present), say(reason), warn(warning)
#
# Measured on 2026-11-22 with `bend <file> -o x.js` (Bend 2.0.40's own JS output, gzip -9; the
# bundler path through bunfig's plugin may differ: slice 3 measures it again):
#   runtime of the Character game (scene + locomotion.frame)   162 KB raw    20 KB gz
#   every project in PROJECT.bend                               218 KB raw    28 KB gz
#   platform/web/*.js                                            25 KB raw   (~8 KB gz, not measured)
#   one library clip (Jog, m1 alone)                            658 KB raw    53 KB gz (above base)
#   all 42 clips (motion(i) names every one)                 35,183 KB raw 2,527 KB gz
#   mannequin assets: skin.bin 259 KB (107 KB gz) + 2 PNGs 1,168 KB  ->  ~1.28 MB on the wire
# Bend's compiler drops defs nothing names: a module that names only Ual.m1() ships one clip even
# though motions.bend is one 3.8 MB file. So a game takes clips by naming them (export/needs.bend),
# never through motion(i).
#
# Limits now:
# - Hashes and sizes are measured by the shell; Bend never reads files (CATCHES X6).
# - Format 1 is web only (Target Web). Native (`app/` instead of `site/`) is format 2.
# - Every Asset is loaded before the first frame: no lazy assets until format 2 (`when` field).
# - Precompressed files never enter the folder: hosts compress (Caddy `encode`).
# - isolation True is refused by the publish host until it can send COOP/COEP per game (publish/service.bend).
#

# ---------- What is in the folder ----------

type Role is Data:
  Page{}          # site/index.html
  Code{}          # site/_/*: JS and CSS the bundler wrote, named by hash
  Asset{}         # site/a/*: files the bundle loads by URL (skin.bin, textures, shaders.json), named by hash

# One served file. `path` relative to site/; `gz` its gzip size; `origin` the repo path it was
# copied from (Assets only; "" for Code and the Page): credits and layers are decided from it.
type Entry is Data:
  Entry{path: String, role: Role, bytes: U32, gz: U32, sha: String, origin: String}

# Third-party work the engine can put inside a game. A closed list: a new collection or character
# is a new case, and the compiler points at every match below.
type Work is Data:
  Mannequin{}     # mannequin/: CC BY 4.0, the credit must be shown (mannequin/credit.bend)
  Quaternius{}    # library/quaternius/: CC0, credit given by courtesy
  BendUi{}        # node_modules/bend-ui/: NO LICENSE FILE today (see RATIONALE, open questions)

type Credit is Data:
  Credit{work: String, author: String, license: String, url: String, notice: String}

# How the game shows its credits: a static page, site/credits.html, written by Bend from
# ui/credits.bend (no Bend-UI JavaScript ships for it). Its presence is MEASURED by the shell,
# never declared: `credits_page` is True only when the file is in the folder. (Graft from
# candidate 2; replaces a declared Shown{InMenu|Hidden}.)

def credit(w: Work) -> Credit:
  match w:
    case Mannequin{}:
      Credit{Mc.title(), Mc.author(), Mc.license(), Mc.url(), Mc.notice()}
    case Quaternius{}:
      Credit{"Universal Animation Library", "Quaternius", "CC0", "https://quaternius.com", ""}
    case BendUi{}:
      Credit{"Bend-UI", "Isabella Herman", "", "https://github.com/isabellaherman/Bend-UI", ""}

# Whether the license requires the credit to be shown (CC BY does; CC0 does not).
def must_show(w: Work) -> Bool:
  match w:
    case Mannequin{}:  True{}
    case Quaternius{}: False{}
    case BendUi{}:     False{}

# The credits a manifest carries: one per work, in the order of `works`. Derived, never written by
# hand, so a work that ships cannot lose its credit on the way to game.json.
def credits(ws: List<&2, Work>) -> List<&2, Credit>:
  match ws:
    case []:
      []
    case w <> rest:
      credit(w) <> credits(rest)

# ---------- Budgets: Code and Assets apart ----------

# What a player waits for before the first frame. All three are measured from the entries.
type Load is Data:
  Load{code_gz: U32, code_raw: U32, asset_wire: U32}

# The reach rule (AGENTS.md: playable by as many people as possible), as numbers:
#   link   200,000 B/s   1.6 Mbit/s, the slow-4G profile of browser audits [MEM]
#   parse  1,000,000 B/s raw JS on a low-end phone [MEM, U: measure on a real device]
# Code must arrive in 2 s and parse in 2 s; Assets must arrive in 8 s.
#   code_gz    300,000   first frame in 1.5 s at 1.6 Mbit/s; Character game today ~28 KB.
#                        One clip through the real page path (Bend loader + Bun, minified, one file
#                        per clip) is ~192 KB raw / ~39 KB gz (judge, measured): ~6 clips fit.
#   code_raw 2,000,000   parse time on a low-end phone; PROVISIONAL, re-measured in slice 3 on the
#                        real minified bundle (the old ~0.66-0.84 MB/clip was unminified `bend -o`):
#                        two clips fit, a third does not (see RATIONALE: clips as data)
#   asset_wire 2,000,000 playable in 10 s at 1.6 Mbit/s; the mannequin is ~1.28 MB (64%)
def link_bytes_per_s() -> U32: 200000
def parse_bytes_per_s() -> U32: 1000000
def budget() -> Load:
  Load{(link_bytes_per_s() * 2 : U32), (parse_bytes_per_s() * 2 : U32), (link_bytes_per_s() * 8 : U32)}

# ---------- The verdict ----------

type Reason is Data:
  ChecksFailed{}
  WrongBend{have: String, want: String}           # export needs the pinned Bend
  ToolMissing{tool: String, shader: String}       # an HLSL shader the game uses, without dxc / spirv-cross
  ShaderNotBuilt{shader: String}                  # in the catalog with diagnostics only
  NoSuchClip{name: String}                        # games.bend names a clip the library does not have
  Forbidden{path: String}                         # an input from a layer that never ships
  NoPage{}
  TwoPages{}
  Unhashed{path: String}
  OverCode{gz: U32, raw: U32, budget: Load}       # Page + Code over budget().code_*
  OverAssets{wire: U32, budget: Load}             # Assets over budget().asset_wire
  Uncredited{work: Work}                          # a work whose license needs its credit shown, with no credits page in the folder

# Printed, never refusing: the owner decides (RATIONALE: mannequin IP; Bend-UI license).
