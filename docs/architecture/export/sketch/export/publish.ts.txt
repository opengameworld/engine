// export/publish.ts: game folder -> a host (SKETCH). Four calls, the same for every host:
//
//   deploy(host, { project })            export, check the folder fits, publish a PREVIEW
//   promote(host, buildId, { actor })    make a published build live (Bend's gate decides)
//   rollback(host, { actor })            promote Hosts.previous(history, live)
//   deploys(host, { project })           the host's history, newest first, live one marked
//
// The host is the single source of truth for what exists and what is live: Vercel's deployment
// list (tagged with --meta ogwBuild=<id>), the VPS's releases/ folder and `live` symlink, itch's
// butler channel. Nothing is recorded locally, so there is no ledger to drift from the host.
//
// Idempotent: publishing a buildId the host already has returns that deployment; promoting the
// live build does nothing; a crash halfway leaves at most an unpromoted upload, which the next
// deploy finds by its buildId.

export type HostName = "vercel" | "vps" | "itch";
export type Actor = "person" | "agent";

// One published build on one host.
export type Deployment = {
  buildId: string;
  url: string;        // its own permanent link (Vercel: the deployment URL; VPS: <id>.preview.<domain> (own origin); itch: the page)
  live: boolean;
  at: string;         // ISO time the host recorded
};

// What every host adapter is (export/hosts/*.ts). Internal to export/: only this file imports
// them (LAYERS: nothing outside export/ imports export/; the fence also forbids hosts/* from
// being imported by anything but publish.ts — one path in, per the red-flag list).
export interface Adapter {
  // Write the host's own files from Bend translations into out/host/<host>/<game>/ and upload.
  // Returns the existing deployment when the host already has this buildId.
  publish(folder: GameFolder, s: Settings): Promise<Deployment>;
  // Point the public name at this build. Only called after Hosts.may_promote said Allowed.
  promote(d: Deployment, s: Settings): Promise<Deployment>;
  // Newest first, from the host itself.
  history(s: Settings): Promise<Deployment[]>;
}
type Settings = { readonly $: "Settings" };   // Bend value from Games.settings(p)

const adapters: Record<HostName, Adapter> = { vercel, vps, itch };
