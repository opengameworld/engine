# Hosts: where a game folder goes, what each host can hold, how the folder's rules are written
# in each host's own config, and who may make a version live. (SKETCH)
#
# A host never decides what is cached or what headers a file gets: contract.bend does. This
# file only TRANSLATES (contract -> vercel config.json, Caddyfile) and states each host's
# LIMITS as numbers, so a folder that cannot fit is refused before anything is uploaded.
#
#   limits(h)               files / bytes per file / bytes in all
#   fits(h, m)              [] or the reasons the folder does not fit h
#   vercel_config(m)        .vercel/output/config.json (Build Output API v3)
#   caddy_site(m, site)     the Caddyfile block for one game on a VPS
#   may_promote(who, m)     only a person promotes, and never a dirty build
#   previous(history, live) the version a rollback goes back to
#
# Limits now:
# - Vercel's per-file static size limit was not found in its docs (2026-11); unknown_bytes()
#   stands for "no limit we know of". Hobby is for non-commercial use.
# - itch has no previews or aliases: a push is live (see promote in export/hosts.ts).
#
# `TODO` marks a body not written yet (sketch).

import Base

type Host is Data:
  Folder{}    # out/web/<game>/ itself, served locally by `bun run export serve`
  Vercel{}    # static CDN; previews, promote, rollback are Vercel's own
  Vps{}       # Caddy on a Linux box we own; previews and live by a folder convention
  Itch{}      # a zip pushed with butler

type Limits is Data:
  Limits{files: U32, file_bytes: U32, total_bytes: U32}

def unknown_bytes() -> U32: 4294967295

def limits(h: Host) -> Limits:
  match h:
    case Folder{}:
      Limits{4294967295, unknown_bytes(), unknown_bytes()}
    case Vercel{}:
      Limits{15000, unknown_bytes(), 104857600}      # 15,000 files per CLI upload; 100 MB on Hobby (1 GB Pro)
    case Vps{}:
      Limits{4294967295, unknown_bytes(), unknown_bytes()}   # the disk is the limit
    case Itch{}:
      Limits{1000, 209715200, 524288000}             # 1,000 files; 200 MB a file; 500 MB in all
