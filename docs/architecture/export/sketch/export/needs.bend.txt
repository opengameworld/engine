# What a game needs: the dependency walk from its roots, and the per-game module export writes
# from it. (SKETCH, new in v2)
#
# Roots: PROJECT.bend's start(p) scene and export/games.bend's settings(p). The walk reads the
# scene as data (engine/scene.bend: things -> look -> figure skin / material shader id) and adds
# the declared clips. The answer, Needs, decides three things:
#   - which skins the page loads (each skin's files become Asset entries: skin.json, skin.bin, textures)
#   - which shader catalog entries the page gets (only these ids; editor/shaders/catalog.generated.json
#     holds every project's)
#   - which library clips are compiled in: the generated module IMPORTS one file per picked clip
#     (library/<c>/gen/m/<slug>.bend). Measured by the judge: through the page path (Bend loader,
#     which exports every def, + Bun) naming one def of motions.bend still ships all 42 clips
#     (9.5 MB raw / 1.93 MB gz). So one file per clip is REQUIRED. That split is in tools/clips.ts
#     (Isabella's area): it goes to her as a request (sketch/requests/), never folded into an
#     export slice. Until it lands, a game with library clips is refused OverCode, honestly.
#
# The per-game module (generated by export/web.ts, untracked, out/gen/<game>/game.bend) is printed
# by the shell from Glue, line for line, as tools/clips.ts prints motions.bend: Bend decides what is
# in it, TS writes it. It exposes what platform/web/run.js's runGame takes:
#   title(), opening(), camera(), clips() -> List<Clip>, credits() -> List<C.Credit>
# and a JS twin, out/gen/<game>/game.js, with the file imports Bun turns into hashed Assets
# (`with {type: "file"}`), as main.js does for the editor today.
#
# Limits now:
# - Skins are found by name (Fig.skin); the files of a skin are known for "mannequin" only
#   (skin_files below). A second character is a new case.
# - A material's shader is found by id; Bend graph materials need no catalog entry.

# A clip the game ships: its collection, its slug (the file library/<c>/gen/m/<slug>.bend) and its name.
type Pick is Data:
  Pick{index: U32, name: String}

type Needs is Data:
  Needs{
    skins: List<&2, String>,      # skin names, deduplicated ("mannequin")
    shaders: List<&2, String>,    # catalog ids of imported materials, deduplicated
    picks: List<&2, Pick>,        # library clips, in declaration order
    unknown: List<&2, String>     # declared clip names the library does not have (-> C.NoSuchClip)
  }

# One file the page imports as a URL (an Asset), and the name the page gives it.
type FileImport is Data:
  FileImport{name: String, path: String}   # e.g. {"mannequin_bin", "mannequin/gen/skin.bin"}

# The generated modules, as data the shell prints.
type Glue is Data:
  Glue{
    project: String,                 # PROJECT.bend constructor to start (folder name)
    picks: List<&2, Pick>,           # -> one `import library/<c>/gen/m/<slug>.bend` line each in out/gen/<game>/clips.bend
    files: List<&2, FileImport>,     # -> one `import <name> from "<path>" with {type:"file"}` each in game.js
    shaders: List<&2, String>        # -> game.js filters the catalog to these ids (written as JSON beside it)
  }
